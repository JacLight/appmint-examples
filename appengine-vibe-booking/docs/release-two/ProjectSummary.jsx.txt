function ProjectSummary({ api, booking, onAuth }) {
  const answers =
    booking.answers &&
    typeof booking.answers === "object" &&
    !Array.isArray(booking.answers)
      ? booking.answers
      : {};
  const invalidAnswers = booking.answers != null && answers !== booking.answers;
  const entries = Object.entries(answers);
  const hasAnswers = entries.length > 0;
  const resource = useRemote(
    () => (hasAnswers ? api.intakeSchema(booking.serviceId) : null),
    [api, booking.serviceId, hasAnswers],
  );
  function answerText(value) {
    if (value == null || value === "") return "Not provided";
    if (typeof value === "boolean") return value ? "Yes" : "No";
    if (typeof value === "string" || typeof value === "number")
      return String(value);
    return "This answer uses an unsupported format.";
  }
  return (
    <section className="project-summary" aria-label="Your project">
      <h3>Your project</h3>
      {invalidAnswers ? (
        <p className="error" role="alert">Saved project answers could not be displayed.</p>
      ) : !hasAnswers ? (
        <p className="muted">No project answers were saved with this appointment.</p>
      ) : resource.loading ? (
        <p className="muted" role="status">Loading your project details…</p>
      ) : (
        <Status resource={resource} onAuth={onAuth}>
          {(schema) => (
            <dl>
              {entries.map(([name, value]) => {
                const title = schema?.properties?.[name]?.title;
                const label =
                  typeof title === "string" && title.trim() ? title : name;
                return (
                  <div key={name}>
                    <dt>{label}</dt>
                    <dd>{answerText(value)}</dd>
                  </div>
                );
              })}
            </dl>
          )}
        </Status>
      )}
    </section>
  );
}
